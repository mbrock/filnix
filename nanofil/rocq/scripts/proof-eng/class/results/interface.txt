Module Type
 Runtime =
 Sig
   Parameter state : Type.
   Parameter start : Nanofil.core.Runtime.Runtime.state.
   Parameter ready :
     Nanofil.core.Runtime.Runtime.state -> Memory.mem -> Prop.
   Parameter alloc :
     Nanofil.core.Runtime.Runtime.state ->
     Memory.alloc_req -> option (Z * Nanofil.core.Runtime.Runtime.state).
   Parameter fresh_allocation :
     forall (s : Nanofil.core.Runtime.Runtime.state)
       (m : Memory.mem) (q : Memory.alloc_req) (base : Z)
       (s' : Nanofil.core.Runtime.Runtime.state),
     Nanofil.core.Runtime.Runtime.ready s m ->
     0 < Memory.req_align q ->
     0 <= Memory.req_extent q ->
     Nanofil.core.Runtime.Runtime.alloc s q = Some (base, s') ->
     Memory.fresh m q base /\
     Nanofil.core.Runtime.Runtime.ready s' (Memory.allocated m q base).
   Parameter address_space :
     forall (s : Nanofil.core.Runtime.Runtime.state) (m : Memory.mem),
     Nanofil.core.Runtime.Runtime.ready s m ->
     objects_fit (Memory.m_regions m).
   Parameter freed_bounds :
     forall (s : Nanofil.core.Runtime.Runtime.state) (m : Memory.mem),
     Nanofil.core.Runtime.Runtime.ready s m ->
     freed_empty (Memory.m_regions m).
   Parameter ready_kept :
     forall (s : Nanofil.core.Runtime.Runtime.state) (m m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.ready s m ->
     kept m m' -> Nanofil.core.Runtime.Runtime.ready s m'.
   Parameter free :
     Values.word -> Memory.mem -> option (Memory.res unit * Memory.mem).
   Parameter free_lifetime :
     forall (x : Values.word) (m : Memory.mem) (r : Memory.res unit)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.free x m = Some (r, m') ->
     lifetime_step m m'.
   Parameter retention :
     forall (x : Values.word) (m : Memory.mem) (r : Memory.res unit)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.free x m = Some (r, m') ->
     Memory.m_bytes m' = Memory.m_bytes m /\ Memory.m_aux m' = Memory.m_aux m.
   Parameter free_keeps_ids :
     forall (x : Values.word) (m : Memory.mem) (r : Memory.res unit)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.free x m = Some (r, m') ->
     forall b : Z,
     Memory.object_region b (Memory.m_regions m) <> None ->
     Memory.object_region b (Memory.m_regions m') <> None.
   Parameter free_checked :
     forall (x : Values.word) (m : Memory.mem) (w : Memory.reason)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.free x m <> Some (Memory.RUnchecked w, m').
   Parameter free_succeeds : frees_heap Nanofil.core.Runtime.Runtime.free.
   Parameter at_safepoint : Syntax.inst -> bool.
   Parameter safepoints :
     forall i : Syntax.inst,
     Nanofil.core.Runtime.Runtime.at_safepoint i = true ->
     Syntax.evidence_preserving i = false.
   Parameter safepoint :
     Nanofil.core.Runtime.Runtime.state ->
     Memory.mem -> Nanofil.core.Runtime.Runtime.state -> Memory.mem -> Prop.
   Parameter collector_live :
     forall (s : Nanofil.core.Runtime.Runtime.state)
       (m : Memory.mem) (s' : Nanofil.core.Runtime.Runtime.state)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.safepoint s m s' m' ->
     Memory.m_regions m' = Memory.m_regions m.
   Parameter safepoint_lifetime :
     forall (s : Nanofil.core.Runtime.Runtime.state)
       (m : Memory.mem) (s' : Nanofil.core.Runtime.Runtime.state)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.safepoint s m s' m' -> lifetime_step m m'.
   Parameter non_moving :
     forall (s : Nanofil.core.Runtime.Runtime.state)
       (m : Memory.mem) (s' : Nanofil.core.Runtime.Runtime.state)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.safepoint s m s' m' ->
     Memory.m_bytes m' = Memory.m_bytes m.
   Parameter aux_round_trip :
     forall (s : Nanofil.core.Runtime.Runtime.state)
       (m : Memory.mem) (s' : Nanofil.core.Runtime.Runtime.state)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.safepoint s m s' m' ->
     forall a : Z,
     Memory.m_aux m' a = Memory.m_aux m a \/
     freed_in (Memory.m_regions m') (Memory.m_aux m a) /\
     traps_as_freed (Memory.m_regions m') (Memory.m_aux m' a).
   Parameter rooting :
     forall (s : Nanofil.core.Runtime.Runtime.state)
       (m : Memory.mem) (s' : Nanofil.core.Runtime.Runtime.state)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.safepoint s m s' m' ->
     forall b : Z,
     Memory.object_region b (Memory.m_regions m) <> None ->
     Memory.object_region b (Memory.m_regions m') <> None.
   Parameter safepoint_ready :
     forall (s : Nanofil.core.Runtime.Runtime.state)
       (m : Memory.mem) (s' : Nanofil.core.Runtime.Runtime.state)
       (m' : Memory.mem),
     Nanofil.core.Runtime.Runtime.safepoint s m s' m' ->
     Nanofil.core.Runtime.Runtime.ready s m ->
     Nanofil.core.Runtime.Runtime.ready s' m'.
 End
Record RuntimeClass : Type := Build_RuntimeClass
  { state : Type;
    start : state;
    ready : state -> Memory.mem -> Prop;
    alloc : state -> Memory.alloc_req -> option (Z * state);
    fresh_allocation : forall (s : state) (m : Memory.mem)
                         (q : Memory.alloc_req) (base : Z)
                         (s' : state),
                       ready s m ->
                       0 < Memory.req_align q ->
                       0 <= Memory.req_extent q ->
                       alloc s q = Some (base, s') ->
                       Memory.fresh m q base /\
                       ready s' (Memory.allocated m q base);
    address_space : forall (s : state) (m : Memory.mem),
                    ready s m -> objects_fit (Memory.m_regions m);
    freed_bounds : forall (s : state) (m : Memory.mem),
                   ready s m -> freed_empty (Memory.m_regions m);
    ready_kept : forall (s : state) (m m' : Memory.mem),
                 ready s m -> kept m m' -> ready s m';
    free : Values.word -> Memory.mem -> option (Memory.res unit * Memory.mem);
    free_lifetime : forall (x : Values.word) (m : Memory.mem)
                      (r : Memory.res unit) (m' : Memory.mem),
                    free x m = Some (r, m') -> lifetime_step m m';
    retention : forall (x : Values.word) (m : Memory.mem)
                  (r : Memory.res unit) (m' : Memory.mem),
                free x m = Some (r, m') ->
                Memory.m_bytes m' = Memory.m_bytes m /\
                Memory.m_aux m' = Memory.m_aux m;
    free_keeps_ids : forall (x : Values.word) (m : Memory.mem)
                       (r : Memory.res unit) (m' : Memory.mem),
                     free x m = Some (r, m') ->
                     forall b : Z,
                     Memory.object_region b (Memory.m_regions m) <> None ->
                     Memory.object_region b (Memory.m_regions m') <> None;
    free_checked : forall (x : Values.word) (m : Memory.mem)
                     (w : Memory.reason) (m' : Memory.mem),
                   free x m <> Some (Memory.RUnchecked w, m');
    free_succeeds : frees_heap free;
    at_safepoint : Syntax.inst -> bool;
    safepoints : forall i : Syntax.inst,
                 at_safepoint i = true ->
                 Syntax.evidence_preserving i = false;
    safepoint : state -> Memory.mem -> state -> Memory.mem -> Prop;
    collector_live : forall (s : state) (m : Memory.mem)
                       (s' : state) (m' : Memory.mem),
                     safepoint s m s' m' ->
                     Memory.m_regions m' = Memory.m_regions m;
    non_moving : forall (s : state) (m : Memory.mem)
                   (s' : state) (m' : Memory.mem),
                 safepoint s m s' m' -> Memory.m_bytes m' = Memory.m_bytes m;
    aux_round_trip : forall (s : state) (m : Memory.mem)
                       (s' : state) (m' : Memory.mem),
                     safepoint s m s' m' ->
                     forall a : Z,
                     Memory.m_aux m' a = Memory.m_aux m a \/
                     freed_in (Memory.m_regions m') (Memory.m_aux m a) /\
                     traps_as_freed (Memory.m_regions m') (Memory.m_aux m' a);
    safepoint_ready : forall (s : state) (m : Memory.mem)
                        (s' : state) (m' : Memory.mem),
                      safepoint s m s' m' -> ready s m -> ready s' m' }.

Arguments R.Build_RuntimeClass state%_type_scope start

  (at_safepoint safepoints safepoint collector_live non_moving aux_round_trip
